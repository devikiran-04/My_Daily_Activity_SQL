--Query to check Billing Process
SELECT PROCESS_STATUS,* FROM CIS.BILLING_CYCLE_PERIOD(NOLOCK) WHERE RECORD_STATUS=1 AND ID=02

SELECT * FROM CIS.HOLIDAY_M(NOLOCK) WHERE YEAR='2022' AND EFFECTIVE_DATE >'2022-12-10 00:00:00' AND RECORD_STATUS=1 ORDER BY EFFECTIVE_DATE ASC

SELECT * FROM CIS.PROCESS_STATUS WHERE ACTIVITY IN ('CONNECT','DISCONNECT','MESSAGESCHEDULER')--,'BILLPROCESSFORMC','READINGEXCEPTION','MONTHLYBILLING')

--//SPM Notifications to check went or not --12Dec 9437, 13Dec 4674,2Jn--13589--13jn-9649--9663
SELECT
COUNT(*) CNT , CONVERT(VARCHAR,CREATE_DATE,112) DATES
FROM CIS.SPM_NOTIFICATIONS (NOLOCK) WHERE RECORD_STATUS=1 AND EVENT_STATUS=1 AND DELIVERY_STATUS='SUCCESS'
AND CREATE_DATE BETWEEN '2022-12-10 00:00:00' AND '2022-12-12 23:59:59'
GROUP BY CONVERT(VARCHAR,CREATE_DATE,112) ORDER BY CONVERT(VARCHAR,CREATE_DATE,112) DESC

--//SPM Connect Disconnect Counts
(SELECT 'TODAY_CONNECTED_COUNT -- ' AS TYPE, COUNT(0) as COUNT FROM CIS.BELOW_ZERO_NOTIFICATION BN WHERE
BN.RECORD_STATUS=1 AND BN.EVENT_STATUS=1 AND BN.METER_STATUS='CONNECT'
AND DATEADD(DD,0,DATEDIFF(DD,0,UPDATE_DATE)) = DATEADD(DD,0,DATEDIFF(DD,0,GETDATE()))) --AS TODAY_CONNECTED_COUNT
UNION ALL
(SELECT 'TODAY_DISCONNECTED_COUNT -- ' AS TYPE,COUNT(0) as COUNT FROM CIS.BELOW_ZERO_NOTIFICATION BN WHERE BN.RECORD_STATUS=1 AND BN.EVENT_STATUS=1 AND BN.METER_STATUS='DISCONNECT'
AND DATEADD(DD,0,DATEDIFF(DD,0,UPDATE_DATE)) = DATEADD(DD,0,DATEDIFF(DD,0,GETDATE()))) --AS TODAY_DISCONNECTED_COUNT
UNION ALL
(SELECT 'PENDING_CONNECT_COUNT -- ' AS TYPE,COUNT(distinct(BN.MTR_NUMBER)) as COUNT
FROM CIS.BELOW_ZERO_NOTIFICATION BN WHERE BN.RECORD_STATUS=1 AND BN.EVENT_STATUS=0 AND BN.METER_STATUS='CONNECT')--AS PENDING_CONNECT_COUNT
UNION ALL
(SELECT 'PENDING_DISCONNECT_COUNT -- ' AS TYPE,COUNT(distinct(BN.MTR_NUMBER)) as COUNT
FROM CIS.BELOW_ZERO_NOTIFICATION BN WHERE BN.RECORD_STATUS=1 AND BN.EVENT_STATUS=0 AND BN.METER_STATUS='DISCONNECT')--AS PENDING_DISCONNECT_COUNT

--//SPM NON CONNECTED METERS AFTER RECHARGE AND ADJUSTMENT POST
SELECT MTR_NUMBER,UPDATE_DATE,* FROM CIS.BELOW_ZERO_NOTIFICATION(NOLOCK) WHERE RECORD_STATUS=1 AND EVENT_STATUS=0 AND
DATEADD(DD,0,DATEDIFF(DD,0,CREATE_DATE)) = DATEADD(DD,0,DATEDIFF(DD,0,GETDATE())) AND METER_STATUS='CONNECT'
ORDER BY CREATE_DATE DESC

---//HES CONNECTED
SELECT METER_NUMBER,COUNT(*) FROM FEP.FEP_ONDEMAND(NOLOCK) WHERE REQUEST_TYPE='DLMSMTRCONDIS'
AND REQ_COMMAND_TYPE='CONN'
AND REQUEST_FROM='SPM_SCHEDULE'
AND CREATE_DATE BETWEEN '2022-10-15 00:00:00.000'
AND '2022-10-15 23:59:31.000' AND RECORD_STATUS=1 AND RESPONSE_CODE='SUCCESSFULLY CONNECTED'
GROUP BY METER_NUMBER 

---//HES DISCONNECTED 
SELECT METER_NUMBER,COUNT(*) FROM FEP.FEP_ONDEMAND(NOLOCK) WHERE REQUEST_TYPE='DLMSMTRCONDIS'
AND REQ_COMMAND_TYPE='DISCONN'
AND REQUEST_FROM='SPM_SCHEDULE'
AND REQUEST_TIME BETWEEN '2022-10-15 00:00:00.000'
AND '2022-10-15 23:59:59' AND RECORD_STATUS=1 AND RESPONSE_CODE='SUCCESSFULLY DISCONNECTED'
GROUP BY METER_NUMBER
