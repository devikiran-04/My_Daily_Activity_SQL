--ED COUNT
--=========
SELECT CDATE,SUM(FEP_ENE_CNT) FEP_ENE_CNT,SUM(MDM_ENE_CNT) MDM_ENE_CNT,sum(MDM_CLONE_ENE_CNT) MDM_CLONE_ENE_CNT FROM (
SELECT CONVERT(VARCHAR,METER_TIME,112) CDATE,COUNT(DISTINCT METER_NUMBER) FEP_ENE_CNT,0 MDM_ENE_CNT,0 MDM_CLONE_ENE_CNT
FROM FEP.FEP_CSV_ED (NOLOCK)
WHERE METER_TIME BETWEEN cast(convert(char(8), GETDATE()-2, 112) + ' 00:00:00' as datetime) AND cast(convert(char(8), GETDATE()-0, 112) + ' 23:59:59' AS DATETIME)
AND status='success'
GROUP BY CONVERT(VARCHAR,METER_TIME,112)
UNION ALL
SELECT BILL_DATE CDATE,0 FEP_ENE_CNT,COUNT(DISTINCT MTR_NUMBER) MDM_ENE_CNT,0 MDM_CLONE_ENE_CNT FROM MDMS.MDM_ENERGY_DATA (NOLOCK)
WHERE BILL_DATE BETWEEN CONVERT(VARCHAR,GETDATE()-2,112) AND CAST(CONVERT(VARCHAR,GETDATE(),112) AS BIGINT) AND IS_RECORD_ESTIMATED=0
AND D_TYPE=1
GROUP BY BILL_DATE
UNION ALL
SELECT BILL_DATE CDATE,0 FEP_ENE_CNT,0 MDM_ENE_CNT,COUNT(DISTINCT MTR_NUMBER) MDM_CLONE_ENE_CNT FROM MDMS.MDM_ENERGY_DATA_CLONE (NOLOCK)
WHERE BILL_DATE BETWEEN CONVERT(VARCHAR,GETDATE()-2,112) AND CAST(CONVERT(VARCHAR,GETDATE(),112) AS BIGINT) AND IS_RECORD_ESTIMATED=0
AND D_TYPE=1
GROUP BY BILL_DATE
) M GROUP BY cdate



--SPM Billing- KESCo
--*****************************
select bill_date,ene_mdm_cnt,bill_cnt,
un_billing_cnt from (SELECT BILL_DATE,
SUM(ENE_ORG_CNT) ENE_MDM_CNT,sum(bill_cnt) bill_cnt,SUM(ENE_ORG_CNT)-sum(bill_cnt) un_billing_cnt
FROM (SELECT BILL_DATE,COUNT(1) ENE_ORG_CNT,0 bill_cnt FROM MDMS.MDM_ENERGY_DATA_CLONE ed (NOLOCK)
inner join mdms.mdm_network_hierarchy nh (nolock) on ed.mtr_number=nh.meter_number and nh.record_status=1 and nh.cons_type='PRE'
WHERE BILL_DATE BETWEEN 20250124 AND 20250126 AND D_TYPE=1 AND IS_RECORD_ESTIMATED=0
GROUP BY BILL_DATE union select CONVERT(VARCHAR,BLISSUE_DT,112) bill_date,0 ENE_ORG_CNT,count(distinct bmeter_num) bill_cnt
from CIS.BILL_DETAILS_T (nolock)
where BLISSUE_DT between '2025-01-24 00:00:00' AND '2025-01-26 23:59:59'
and record_status=1
group by CONVERT(VARCHAR,BLISSUE_DT,112)
) M GROUP BY BILL_DATE
) Z
order by bill_date asc
